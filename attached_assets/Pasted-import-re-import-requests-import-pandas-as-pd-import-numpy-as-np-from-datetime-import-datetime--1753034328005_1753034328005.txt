import re
import requests
import pandas as pd
import numpy as np
from datetime import datetime, timedelta
import time
import json
from collections import defaultdict
from typing import Dict, List, Tuple, Optional
import os

class TradePlan:
    """Represents a single trade plan"""
    def __init__(self):
        self.symbol = None
        self.bias = None
        self.strike = None
        self.option_type = None
        self.expiry = None
        self.entry_price = None
        self.stop_loss = None
        self.initial_target = None
        self.final_target = None
        self.position_size = None
        self.max_hold_time = None
        self.delta = None
        self.gamma = None
        self.theta = None
        self.score = None
        
    def __repr__(self):
        return f"TradePlan({self.symbol} ${self.strike} {self.option_type} exp:{self.expiry})"

class OptionsBacktester:
    def __init__(self, api_key: str, trades_file: str):
        self.api_key = api_key
        self.trades_file = trades_file
        self.trades = []
        self.unique_options = set()
        self.historical_data = {}
        self.results = []
        
    def parse_trades_file(self):
        """Parse the trade plans from the text file"""
        print(f"Parsing trades from {self.trades_file}...")
        
        with open(self.trades_file, 'r', encoding='utf-8') as f:
            content = f.read()
        
        # Split by trade plan sections
        trade_sections = content.split('📈 Trade Plan for')
        
        for section in trade_sections[1:]:  # Skip first empty section
            trade = TradePlan()
            
            # Extract symbol and bias
            symbol_match = re.search(r'(\w+)\s+\((\w+)\s+Bias\)', section)
            if symbol_match:
                trade.symbol = symbol_match.group(1)
                trade.bias = symbol_match.group(2)
            
            # Extract option details
            option_match = re.search(r'\$\$?([\d.]+)\s+(Call|Put)\s+expiring\s+([\d-]+)', section)
            if option_match:
                trade.strike = float(option_match.group(1))
                trade.option_type = option_match.group(2)
                trade.expiry = option_match.group(3)
            
            # Extract prices
            entry_match = re.search(r'Entry Price:\s*\$?([\d.]+)', section)
            if entry_match:
                trade.entry_price = float(entry_match.group(1))
            
            stop_match = re.search(r'Stop Loss:\s*\$?([\d.]+)', section)
            if stop_match:
                trade.stop_loss = float(stop_match.group(1))
                
            initial_match = re.search(r'Initial Target:\s*\$?([\d.]+)', section)
            if initial_match:
                trade.initial_target = float(initial_match.group(1))
                
            final_match = re.search(r'Final Target:\s*\$?([\d.]+)', section)
            if final_match:
                trade.final_target = float(final_match.group(1))
            
            # Extract position size and hold time
            size_match = re.search(r'Position Size:\s*(\d+)', section)
            if size_match:
                trade.position_size = int(size_match.group(1))
                
            hold_match = re.search(r'Max Hold Time:\s*(\d+)\s*hours?', section)
            if hold_match:
                trade.max_hold_time = int(hold_match.group(1))
            
            # Extract Greeks
            metrics_match = re.search(r'Delta:\s*([-\d.]+).*?Gamma:\s*([\d.]+).*?Theta:\s*([-\d.]+).*?Score:\s*([\d.]+)', section)
            if metrics_match:
                trade.delta = float(metrics_match.group(1))
                trade.gamma = float(metrics_match.group(2))
                trade.theta = float(metrics_match.group(3))
                trade.score = float(metrics_match.group(4))
            
            if trade.symbol and trade.strike and trade.option_type and trade.expiry:
                self.trades.append(trade)
                # Create unique option identifier
                option_id = (trade.symbol, trade.strike, trade.option_type, trade.expiry)
                self.unique_options.add(option_id)
        
        print(f"Parsed {len(self.trades)} trades with {len(self.unique_options)} unique options")
        
    def fetch_historical_data(self, symbol: str, date: str) -> Optional[pd.DataFrame]:
        """Fetch historical options data from Alpha Vantage"""
        url = f'https://www.alphavantage.co/query?function=HISTORICAL_OPTIONS&symbol={symbol}&date={date}&apikey={self.api_key}'
        
        try:
            response = requests.get(url)
            data = response.json()
            
            if 'data' in data:
                df = pd.DataFrame(data['data'])
                return df
            else:
                print(f"No data found for {symbol} on {date}")
                return None
                
        except Exception as e:
            print(f"Error fetching data for {symbol} on {date}: {e}")
            return None
    
    def find_option_prices(self, df: pd.DataFrame, strike: float, option_type: str, 
                          expiry: str, start_time: datetime, hours: int = 12) -> Dict:
        """Find price movements for a specific option over the specified timeframe"""
        # Filter for the specific option
        option_df = df[
            (df['strike'] == strike) & 
            (df['type'] == option_type.upper()) & 
            (df['expiration'] == expiry)
        ]
        
        if option_df.empty:
            return None
        
        results = {
            'entry_price': None,
            'max_price': None,
            'min_price': None,
            'exit_price': None,
            'hit_times': {}
        }
        
        # For historical data, we'll use mid price (average of bid/ask)
        if 'bid' in option_df.columns and 'ask' in option_df.columns:
            option_df['mid_price'] = (option_df['bid'].astype(float) + option_df['ask'].astype(float)) / 2
        else:
            # If no bid/ask, use last price
            option_df['mid_price'] = option_df['last'].astype(float)
        
        # Since we don't have intraday data, we'll simulate with available data
        # This is a limitation - Alpha Vantage historical data is daily
        results['entry_price'] = option_df['mid_price'].iloc[0] if not option_df.empty else None
        results['max_price'] = option_df['mid_price'].max() if not option_df.empty else None
        results['min_price'] = option_df['mid_price'].min() if not option_df.empty else None
        results['exit_price'] = option_df['mid_price'].iloc[-1] if not option_df.empty else None
        
        return results
    
    def backtest_all_trades(self):
        """Backtest all parsed trades"""
        print("\nStarting backtesting process...")
        
        # Group trades by symbol and date for efficient API calls
        trades_by_symbol_date = defaultdict(list)
        
        for trade in self.trades:
            # We need to determine what date to fetch data for
            # Since these are future dates, we'll need to adjust
            # For now, let's use the current date minus the time to expiry
            expiry_date = datetime.strptime(trade.expiry, '%Y-%m-%d')
            days_to_expiry = (expiry_date - datetime.now()).days
            
            # Fetch data for multiple dates to simulate the trade
            # We'll fetch data for the entry date (estimated)
            entry_date = datetime.now() - timedelta(days=30)  # Assume trades were placed 30 days ago
            
            trades_by_symbol_date[(trade.symbol, entry_date.strftime('%Y-%m-%d'))].append(trade)
        
        # Fetch data and backtest
        for (symbol, date), trades in trades_by_symbol_date.items():
            print(f"\nFetching data for {symbol} on {date}...")
            
            # Rate limiting - Alpha Vantage free tier allows 5 calls/minute
            time.sleep(12)  # Wait 12 seconds between calls
            
            df = self.fetch_historical_data(symbol, date)
            
            if df is not None:
                for trade in trades:
                    result = self.backtest_single_trade(trade, df)
                    self.results.append(result)
            else:
                # Mark trades as unable to backtest
                for trade in trades:
                    self.results.append({
                        'trade': trade,
                        'status': 'NO_DATA',
                        'profit_loss': 0,
                        'profit_loss_pct': 0,
                        'hit_initial': False,
                        'hit_final': False,
                        'hit_stop': False
                    })
    
    def backtest_single_trade(self, trade: TradePlan, df: pd.DataFrame) -> Dict:
        """Backtest a single trade"""
        # Find the option in the dataframe
        option_data = self.find_option_prices(
            df, 
            trade.strike, 
            trade.option_type,
            trade.expiry,
            datetime.now(),  # This should be the actual trade entry time
            trade.max_hold_time
        )
        
        if not option_data or option_data['entry_price'] is None:
            return {
                'trade': trade,
                'status': 'OPTION_NOT_FOUND',
                'profit_loss': 0,
                'profit_loss_pct': 0,
                'hit_initial': False,
                'hit_final': False,
                'hit_stop': False
            }
        
        # Calculate performance
        actual_entry = option_data['entry_price']
        max_price = option_data['max_price']
        min_price = option_data['min_price']
        
        # Check if targets or stop loss were hit
        hit_initial = max_price >= trade.initial_target if max_price else False
        hit_final = max_price >= trade.final_target if max_price else False
        hit_stop = min_price <= trade.stop_loss if min_price else False
        
        # Calculate profit/loss
        if hit_stop:
            exit_price = trade.stop_loss
            status = 'STOPPED_OUT'
        elif hit_final:
            exit_price = trade.final_target
            status = 'FINAL_TARGET'
        elif hit_initial:
            exit_price = trade.initial_target
            status = 'INITIAL_TARGET'
        else:
            exit_price = option_data['exit_price']
            status = 'TIME_EXIT'
        
        profit_loss = (exit_price - trade.entry_price) * trade.position_size * 100
        profit_loss_pct = ((exit_price - trade.entry_price) / trade.entry_price) * 100
        
        return {
            'trade': trade,
            'status': status,
            'profit_loss': profit_loss,
            'profit_loss_pct': profit_loss_pct,
            'hit_initial': hit_initial,
            'hit_final': hit_final,
            'hit_stop': hit_stop,
            'actual_entry': actual_entry,
            'expected_entry': trade.entry_price,
            'max_price': max_price,
            'min_price': min_price
        }
    
    def analyze_results(self):
        """Analyze the backtest results to find patterns"""
        print("\n" + "="*80)
        print("BACKTEST RESULTS ANALYSIS")
        print("="*80)
        
        # Convert results to DataFrame for easier analysis
        results_data = []
        for r in self.results:
            trade = r['trade']
            results_data.append({
                'symbol': trade.symbol,
                'bias': trade.bias,
                'strike': trade.strike,
                'type': trade.option_type,
                'status': r['status'],
                'profit_loss': r['profit_loss'],
                'profit_loss_pct': r['profit_loss_pct'],
                'hit_initial': r['hit_initial'],
                'hit_final': r['hit_final'],
                'hit_stop': r['hit_stop'],
                'delta': trade.delta,
                'gamma': trade.gamma,
                'theta': trade.theta,
                'score': trade.score
            })
        
        df_results = pd.DataFrame(results_data)
        
        # Overall statistics
        print("\n1. OVERALL PERFORMANCE:")
        print(f"   Total trades analyzed: {len(df_results)}")
        print(f"   Trades with data: {len(df_results[df_results['status'] != 'NO_DATA'])}")
        
        # Success rates
        if len(df_results[df_results['status'] != 'NO_DATA']) > 0:
            valid_trades = df_results[df_results['status'] != 'NO_DATA']
            print(f"\n2. SUCCESS RATES:")
            print(f"   Hit initial target: {valid_trades['hit_initial'].sum()} ({valid_trades['hit_initial'].mean()*100:.1f}%)")
            print(f"   Hit final target: {valid_trades['hit_final'].sum()} ({valid_trades['hit_final'].mean()*100:.1f}%)")
            print(f"   Hit stop loss: {valid_trades['hit_stop'].sum()} ({valid_trades['hit_stop'].mean()*100:.1f}%)")
            
            # Profitability
            print(f"\n3. PROFITABILITY:")
            print(f"   Average P&L: ${valid_trades['profit_loss'].mean():.2f}")
            print(f"   Average P&L %: {valid_trades['profit_loss_pct'].mean():.2f}%")
            print(f"   Total P&L: ${valid_trades['profit_loss'].sum():.2f}")
            
            # Analyze by Greeks for successful trades
            successful_trades = valid_trades[valid_trades['hit_initial'] | valid_trades['hit_final']]
            
            if len(successful_trades) > 0:
                print(f"\n4. CHARACTERISTICS OF SUCCESSFUL TRADES:")
                print(f"   Average Delta: {successful_trades['delta'].mean():.4f}")
                print(f"   Average Gamma: {successful_trades['gamma'].mean():.4f}")
                print(f"   Average Theta: {successful_trades['theta'].mean():.4f}")
                print(f"   Average Score: {successful_trades['score'].mean():.2f}")
                
                # Compare to unsuccessful trades
                unsuccessful_trades = valid_trades[~(valid_trades['hit_initial'] | valid_trades['hit_final'])]
                if len(unsuccessful_trades) > 0:
                    print(f"\n5. SUCCESSFUL vs UNSUCCESSFUL TRADES:")
                    print(f"   Delta: {successful_trades['delta'].mean():.4f} vs {unsuccessful_trades['delta'].mean():.4f}")
                    print(f"   Gamma: {successful_trades['gamma'].mean():.4f} vs {unsuccessful_trades['gamma'].mean():.4f}")
                    print(f"   Score: {successful_trades['score'].mean():.2f} vs {unsuccessful_trades['score'].mean():.2f}")
            
            # Best performing patterns
            print(f"\n6. BEST PERFORMING PATTERNS:")
            
            # Group by score ranges
            score_bins = [0, 50, 75, 100, 150]
            df_results['score_range'] = pd.cut(df_results['score'], bins=score_bins)
            score_analysis = df_results.groupby('score_range').agg({
                'hit_initial': 'mean',
                'hit_final': 'mean',
                'profit_loss_pct': 'mean'
            })
            print("\n   Success rate by score range:")
            print(score_analysis)
            
        # Save detailed results
        self.save_results(df_results)
    
    def save_results(self, df_results: pd.DataFrame):
        """Save results to CSV for further analysis"""
        output_file = 'backtest_results.csv'
        df_results.to_csv(output_file, index=False)
        print(f"\nDetailed results saved to {output_file}")
        
        # Also save a summary JSON
        summary = {
            'total_trades': len(self.trades),
            'unique_options': len(self.unique_options),
            'trades_with_data': len(df_results[df_results['status'] != 'NO_DATA']),
            'overall_profit_loss': float(df_results['profit_loss'].sum()),
            'success_rate_initial': float(df_results['hit_initial'].mean() * 100) if len(df_results) > 0 else 0,
            'success_rate_final': float(df_results['hit_final'].mean() * 100) if len(df_results) > 0 else 0,
            'timestamp': datetime.now().isoformat()
        }
        
        with open('backtest_summary.json', 'w') as f:
            json.dump(summary, f, indent=2)
        print(f"Summary saved to backtest_summary.json")

def main():
    # Configuration
    API_KEY = 'YOUR_ALPHA_VANTAGE_API_KEY'  # Replace with your actual API key
    TRADES_FILE = 'trades.txt'  # Your 14k line file
    
    # Initialize backtester
    backtester = OptionsBacktester(API_KEY, TRADES_FILE)
    
    # Run the backtest
    backtester.parse_trades_file()
    backtester.backtest_all_trades()
    backtester.analyze_results()
    
    print("\nBacktesting complete!")

if __name__ == "__main__":
    main()
